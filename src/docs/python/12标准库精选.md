## 一、pathlib：现代文件路径操作

### 1.1 Path 对象

```python
from pathlib import Path

# 创建路径
p = Path("docs/python/01index.md")
print(p)             # docs/python/01index.md
print(p.name)        # 01index.md
print(p.stem)        # 01index（不含扩展名）
print(p.suffix)      # .md
print(p.parent)      # docs/python
print(p.parts)       # ('docs', 'python', '01index.md')

# 绝对路径
abs_path = p.resolve()
print(abs_path)      # /full/path/to/docs/python/01index.md

# 相对路径
rel = abs_path.relative_to(Path.cwd())
print(rel)           # docs/python/01index.md
```

### 1.2 文件操作

```python
from pathlib import Path

# 检查
p = Path("docs/python/01index.md")
print(p.exists())    # True
print(p.is_file())   # True
print(p.is_dir())    # False

# 读取/写入
content = p.read_text(encoding="utf-8")
p.write_text("# New content", encoding="utf-8")

# 追加
p.write_text("\n## Added line", encoding="utf-8", mode="a")

# 创建目录
Path("new_dir").mkdir(exist_ok=True)
Path("a/b/c").mkdir(parents=True, exist_ok=True)  # 递归创建

# 删除
p.unlink()           # 删除文件
Path("empty_dir").rmdir()  # 删除空目录

# 列举文件
for md_file in Path("docs").glob("*.md"):
    print(md_file)

# 递归列举
for md_file in Path("docs").rglob("**/*.md"):
    print(md_file)
```

### 1.3 AI/LLM 场景：批量处理 Markdown 文件

```python
from pathlib import Path

def collect_tutorial_files(root_dir="docs/tutorial"):
    """收集教程目录下所有 .md 文件"""
    path = Path(root_dir)
    files = sorted(path.glob("*.md"))
    return [f for f in files if f.is_file() and f.name != "index.md"]

def read_all_titles(files):
    """读取所有文件的标题"""
    titles = {}
    for f in files:
        content = f.read_text(encoding="utf-8")
        # 简单提取 YAML front matter 中的 title
        if content.startswith("---"):
            parts = content.split("---", 2)
            if len(parts) >= 2:
                for line in parts[1].split("\n"):
                    if line.startswith("title:"):
                        titles[f.stem] = line.split(":", 1)[1].strip()
                        break
    return titles

files = collect_tutorial_files()
titles = read_all_titles(files)
for name, title in titles.items():
    print(f"{name}: {title}")
```

## 二、datetime：日期时间处理

### 2.1 基本操作

```python
from datetime import datetime, date, time, timedelta

# 当前时间
now = datetime.now()
today = date.today()
print(now)           # 2024-01-15 14:30:00.123456
print(today)         # 2024-01-15

# 创建指定时间
dt = datetime(2024, 1, 15, 14, 30, 0)
print(dt)            # 2024-01-15 14:30:00

# 格式化
print(now.strftime("%Y-%m-%d %H:%M:%S"))  # 2024-01-15 14:30:00
print(now.strftime("%B %d, %Y"))           # January 15, 2024

# 解析
parsed = datetime.strptime("2024-01-15", "%Y-%m-%d")
print(parsed)  # 2024-01-15 00:00:00

# 时间差
delta = timedelta(days=7, hours=3)
future = now + delta
past = now - delta

# 计算年龄
birthday = date(1999, 6, 15)
age = (today - birthday).days // 365
print(f"年龄：{age} 岁")
```

### 2.2 时区处理

```python
from datetime import datetime, timezone, timedelta

# UTC 时间
utc_now = datetime.now(timezone.utc)
print(utc_now)  # 2024-01-15 06:30:00+00:00

# 北京时间 (UTC+8)
beijing_tz = timezone(timedelta(hours=8))
beijing_now = datetime.now(beijing_tz)
print(beijing_now)  # 2024-01-15 14:30:00+08:00

# 时区转换
utc_str = "2024-01-15T06:30:00+00:00"
dt = datetime.fromisoformat(utc_str)
beijing_dt = dt.astimezone(beijing_tz)
print(beijing_dt)  # 2024-01-15 14:30:00+08:00
```

## 三、collections：专用数据结构

### 3.1 namedtuple：轻量级记录

```python
from collections import namedtuple

# 定义命名元组
Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)
print(p.x)       # 3
print(p.y)       # 4
print(p[0])      # 3（也支持下标）

# AI 场景：定义 Token 结构
Token = namedtuple("Token", ["text", "start", "end", "type"])
t = Token("Hello", 0, 5, "word")
print(t.text)  # Hello
```

### 3.2 deque：双端队列

```python
from collections import deque

# 高效的两端操作
queue = deque(["task1", "task2", "task3"])
queue.append("task4")       # 右端添加
queue.appendleft("task0")   # 左端添加
first = queue.pop()         # 右端弹出
last = queue.popleft()      # 左端弹出

# 滑动窗口（固定大小）
def sliding_window(iterable, n):
    """返回大小为 n 的滑动窗口"""
    window = deque(maxlen=n)
    for item in iterable:
        window.append(item)
        yield tuple(window)

for w in sliding_window(range(5), 3):
    print(w)
# (0, 1, 2)
# (1, 2, 3)
# (2, 3, 4)
```

### 3.3 defaultdict 与 Counter

```python
from collections import defaultdict, Counter

# defaultdict：自动创建默认值
groups = defaultdict(list)
for name, group in [("Alice", "A"), ("Bob", "B"), ("Charlie", "A")]:
    groups[group].append(name)
print(dict(groups))
# {'A': ['Alice', 'Charlie'], 'B': ['Bob']}

# Counter：计数字典
words = ["apple", "banana", "apple", "orange", "banana", "apple"]
counts = Counter(words)
print(counts.most_common(2))  # [('apple', 3), ('banana', 2)]
print(counts["pear"])         # 0（不存在的键返回 0）

# Counter 运算
c1 = Counter(["a", "b", "c", "a"])
c2 = Counter(["a", "a", "b"])
print(c1 + c2)    # Counter({'a': 4, 'b': 2, 'c': 1})
print(c1 - c2)    # Counter({'c': 1})（减法：取正值）
print(c1 & c2)    # Counter({'a': 2, 'b': 1})（交集）
print(c1 | c2)    # Counter({'a': 4, 'b': 2, 'c': 1})（并集）
```

## 五、shutil：高级文件操作

```python
import shutil
from pathlib import Path

# 复制文件
shutil.copy("source.md", "backup.md")
shutil.copy2("source.md", "backup.md")  # 保留元数据

# 复制目录树
shutil.copytree("src_dir", "dst_dir")

# 移动文件/目录
shutil.move("old_name.md", "new_name.md")

# 删除目录树
shutil.rmtree("temp_dir")

# 压缩/解压
shutil.make_archive("archive", "zip", "root_dir")
shutil.unpack_archive("archive.zip", "extract_dir")

# 大文件复制（分块）
def copy_large(src, dst, chunk_size=8192):
    with open(src, "rb") as fsrc, open(dst, "wb") as fdst:
        while chunk := fsrc.read(chunk_size):
            fdst.write(chunk)
```

## 六、hashlib：哈希算法

```python
import hashlib

# MD5（不推荐用于安全场景）
md5 = hashlib.md5(b"hello").hexdigest()
print(md5)  # 5d41402abc4b2a76b9719d911017c592

# SHA-256（推荐）
sha256 = hashlib.sha256(b"hello").hexdigest()
print(sha256)  # 2cf24dba5fb0a30e26e83b2ac5b9e29e1b161e5c1fa7425e73043362938b9824

# 大文件哈希（分块计算）
def file_hash(filepath, algorithm="sha256", chunk_size=8192):
    h = hashlib.new(algorithm)
    with open(filepath, "rb") as f:
        while chunk := f.read(chunk_size):
            h.update(chunk)
    return h.hexdigest()

print(file_hash("large_file.bin"))
```

### 6.1 AI/LLM 场景：文本指纹

```python
import hashlib

def text_fingerprint(text: str) -> str:
    """生成文本的 SHA-256 指纹，用于去重"""
    normalized = text.strip().lower()
    return hashlib.sha256(normalized.encode("utf-8")).hexdigest()[:16]

# 去重 LLM 回复
responses = [
    "Hello, how can I help you?",
    "hello, how can i help you?",  # 去重后视为相同
    "What's the weather today?",
]

seen = set()
unique_responses = []
for r in responses:
    fp = text_fingerprint(r)
    if fp not in seen:
        seen.add(fp)
        unique_responses.append(r)

print(unique_responses)
# ['Hello, how can I help you?', "What's the weather today?"]
```
