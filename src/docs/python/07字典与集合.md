## 一、字典：键值对的映射

### 1.1 创建与访问

```python
# 多种方式创建字典
person = {"name": "Alice", "age": 25, "city": "北京"}
empty = {}
empty2 = dict()
from_pairs = dict([("name", "Alice"), ("age", 25)])
from_keys = dict.fromkeys(["name", "age", "city"], None)

# 访问值
print(person["name"])      # Alice
print(person.get("phone")) # None（键不存在时返回 None）
print(person.get("phone", "010-1234567"))  # 默认值

# 注意区别
# person["phone"]   # ✗ KeyError：键不存在时报错
# person.get("phone")  # None：键不存在时安全返回 None
```

> **`[]` vs `.get()`**：`[]` 在键不存在时抛出 KeyError，`.get()` 返回 None 或指定默认值。不确定键是否存在时用 `.get()`。

### 1.2 增删改

```python
person = {"name": "Alice", "age": 25}

# 新增/修改
person["email"] = "alice@example.com"  # 新增
person["age"] = 26                      # 修改
person.setdefault("phone", "010-1234")  # 键不存在时设置默认值

# 删除
del person["email"]
removed = person.pop("age")       # 删除并返回值
removed = person.popitem()        # 删除并返回最后一对（Python 3.7+ 有序）
# person.clear()                  # 清空

# 批量更新
person.update({"age": 27, "city": "上海"})
```

### 1.3 遍历

```python
person = {"name": "Alice", "age": 25, "city": "北京"}

# 遍历键（默认）
for key in person:
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")

# 遍历前 N 个
for i, (key, value) in enumerate(list(person.items())[:2]):
    print(f"{i}. {key}: {value}")
```

### 1.4 嵌套字典

```python
# 多层嵌套
users = {
    "alice": {
        "name": "Alice Wang",
        "scores": {"math": 95, "english": 88},
        "contacts": {"email": "alice@example.com", "phone": "123"}
    },
    "bob": {
        "name": "Bob Li",
        "scores": {"math": 82, "english": 91},
        "contacts": {"email": "bob@example.com"}
    }
}

# 访问深层嵌套
print(users["alice"]["scores"]["math"])  # 95

# 安全访问（避免 KeyError）
email = users.get("alice", {}).get("contacts", {}).get("email", "未知")

# 遍历所有用户的数学成绩
for username, info in users.items():
    math_score = info.get("scores", {}).get("math", 0)
    print(f"{username}: 数学 {math_score} 分")
```

### 1.5 字典推导式

```python
# 基本推导
squares = {x: x**2 for x in range(6)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# 条件过滤
evens = {x: x**2 for x in range(10) if x % 2 == 0}
# {0: 0, 2: 4, 4: 16, 6: 36, 8: 64}

# 反转键值
mapping = {"a": 1, "b": 2, "c": 3}
reversed_map = {v: k for k, v in mapping.items()}
# {1: 'a', 2: 'b', 3: 'c'}

# 统计词频
words = ["apple", "banana", "apple", "orange", "banana", "apple"]
freq = {w: words.count(w) for w in set(words)}
# {'apple': 3, 'banana': 2, 'orange': 1}
```

## 二、字典的高级特性

### 2.1 collections 模块

```python
from collections import defaultdict, Counter, OrderedDict

# defaultdict：键不存在时自动创建默认值
word_freq = defaultdict(int)
words = ["apple", "banana", "apple", "orange", "banana", "apple"]
for w in words:
    word_freq[w] += 1
print(dict(word_freq))  # {'apple': 3, 'banana': 2, 'orange': 1}

# 不用 defaultdict 的传统写法（更繁琐）
word_freq2 = {}
for w in words:
    if w not in word_freq2:
        word_freq2[w] = 0
    word_freq2[w] += 1

# Counter：计数字典
counts = Counter(words)
print(counts)  # Counter({'apple': 3, 'banana': 2, 'orange': 1})
print(counts.most_common(2))  # [('apple', 3), ('banana', 2)]

# 合并两个 Counter
c1 = Counter(["a", "b", "c"])
c2 = Counter(["b", "c", "d"])
print(c1 + c2)  # Counter({'b': 2, 'c': 2, 'a': 1, 'd': 1})
```

### 2.2 字典合并（Python 3.9+）

```python
dict1 = {"a": 1, "b": 2}
dict2 = {"b": 3, "c": 4}

# 合并运算符（Python 3.9+）
merged = dict1 | dict2  # {"a": 1, "b": 3, "c": 4}（dict2 覆盖 dict1）
dict1 |= dict2          # 就地合并

# 旧写法（Python 3.5+）
merged2 = {**dict1, **dict2}
```

## 三、集合：无序的唯一元素

### 3.1 创建与基本操作

```python
# 创建集合
fruits = {"苹果", "香蕉", "橙子"}
empty = set()  # 注意：{} 是空字典，不是空集合！
from_list = set([1, 2, 2, 3, 3, 3])  # {1, 2, 3}（去重）

# 增删
fruits.add("葡萄")
fruits.update(["芒果", "火龙果"])  # 批量添加
fruits.remove("苹果")              # 不存在则抛 KeyError
fruits.discard("苹果")             # 不存在则静默（推荐）
removed = fruits.pop()             # 随机移除并返回一个元素

# 成员检查（O(1) 复杂度）
print("香蕉" in fruits)  # True
```

### 3.2 集合运算

```python
a = {1, 2, 3, 4, 5}
b = {4, 5, 6, 7, 8}

# 并集
print(a | b)         # {1, 2, 3, 4, 5, 6, 7, 8}
print(a.union(b))    # 同上

# 交集
print(a & b)         # {4, 5}
print(a.intersection(b))

# 差集
print(a - b)         # {1, 2, 3}（在 a 但不在 b）
print(b - a)         # {6, 7, 8}

# 对称差集（在 a 或 b 中，但不同时在两者中）
print(a ^ b)         # {1, 2, 3, 6, 7, 8}

# 子集/超集
print({1, 2} <= {1, 2, 3})   # True（子集）
print({1, 2, 3} >= {1, 2})   # True（超集）
```

### 3.3 集合推导式

```python
# 去重
words = ["apple", "banana", "apple", "orange", "banana"]
unique = {w for w in words}
print(unique)  # {'apple', 'banana', 'orange'}

# 转换
upper_words = {w.upper() for w in words}
print(upper_words)  # {'APPLE', 'BANANA', 'ORANGE'}
```

## 四、字典 vs 集合 vs 列表

| 数据结构 | 有序 | 可变 | 唯一 | 查找速度 | 典型用途 |
|---------|------|------|------|---------|---------|
| 列表 `[]` | ✓ | ✓ | ✗ | O(n) | 有序序列、队列 |
| 元组 `()` | ✓ | ✗ | ✗ | O(n) | 固定记录、字典键 |
| 字典 `{k:v}` | ✓ | ✓ | 键唯一 | O(1) | 键值映射、配置 |
| 集合 `{}` | ✗ | ✓ | ✓ | O(1) | 去重、成员检查 |

## 五、AI/LLM 场景实战

### 5.1 构建 API 请求体

```python
# OpenAI Chat API 的请求体
request_body = {
    "model": "gpt-4",
    "messages": [
        {"role": "system", "content": "你是一个有用的助手。"},
        {"role": "user", "content": "Python 列表和元组有什么区别？"},
    ],
    "temperature": 0.7,
    "max_tokens": 500,
}

# 动态构建
def build_request(messages, model="gpt-4", **kwargs):
    """构建 API 请求体，kwargs 中的参数会覆盖默认值"""
    request = {
        "model": model,
        "messages": messages,
        **kwargs,  # 展开额外参数
    }
    return request

req = build_request(
    [{"role": "user", "content": "你好"}],
    temperature=0.5,
    top_p=0.9,
)
```

### 5.2 解析 API 响应

```python
# 模拟 API 响应
response = {
    "choices": [
        {
            "message": {
                "role": "assistant",
                "content": "Python 列表和元组的主要区别在于可变性。"
            },
            "finish_reason": "stop"
        }
    ],
    "usage": {
        "prompt_tokens": 15,
        "completion_tokens": 20,
        "total_tokens": 35
    }
}

# 安全提取
choice = response.get("choices", [{}])[0]
message = choice.get("message", {})
content = message.get("content", "")
print(content)  # Python 列表和元组的主要区别在于可变性。

# 统计 token 使用情况
usage = response.get("usage", {})
print(f"消耗 tokens: {usage.get('total_tokens', 0)}")
```

### 5.3 词频统计与去重

```python
from collections import Counter

# 文本分词（简化版：按空格分割）
text = "the cat sat on the mat the cat"
words = text.split()

# 词频统计
word_counts = Counter(words)
print(word_counts.most_common(3))  # [('the', 3), ('cat', 2), ('sat', 1)]

# 去重
unique_words = set(words)
print(unique_words)  # {'the', 'cat', 'sat', 'on', 'mat'}

# 找出只出现一次的词
rare_words = {w for w, c in word_counts.items() if c == 1}
print(rare_words)  # {'sat', 'on', 'mat'}
```

## 六、本章小结

| 概念 | 核心要点 |
|------|---------|
| 字典创建 | `{"key": value}` 或 `dict()` |
| 安全访问 | `.get(key, default)` 避免 KeyError |
| 遍历 | `.items()` / `.keys()` / `.values()` |
| defaultdict | 键不存在时自动创建默认值 |
| Counter | 计数字典，`.most_common(n)` |
| 集合 | `{}` 创建，去重、成员检查 O(1) |
| 集合运算 | `|` 并集 `&` 交集 `-` 差集 `^` 对称差 |
| 字典合并 | `dict1 \| dict2`（Python 3.9+） |
