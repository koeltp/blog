## 一、装饰器原理

### 1.1 函数是一等公民

```python
# Python 中函数可以像普通变量一样传递
def greet(name):
    return f"Hello, {name}!"

# 赋值
say_hello = greet
print(say_hello("Alice"))  # Hello, Alice!

# 作为参数传递
def call_twice(func, arg):
    print(func(arg))
    print(func(arg))

call_twice(greet, "Bob")

# 在函数内定义函数
def outer():
    def inner():
        return "I'm inner"
    return inner

fn = outer()
print(fn())  # I'm inner
```

> **理解这一点是理解装饰器的关键**：函数可以返回函数、可以作为参数传递。装饰器本质上就是一个"接收函数、返回函数"的函数。

### 1.2 手写装饰器

```python
import time
from functools import wraps

# 不带参数的装饰器
def timer(func):
    """测量函数执行时间的装饰器"""
    @wraps(func)  # 保留原函数的 __name__ 和 __doc__
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} 耗时 {elapsed:.4f}s")
        return result
    return wrapper

# 使用
@timer
def slow_sum(n):
    """计算 1 到 n 的和"""
    return sum(range(n))

slow_sum(1_000_000)
# slow_sum 耗时 0.0456s
# （同时保留了 slow_sum.__doc__ == "计算 1 到 n 的和"）
```

> **`@wraps` 的作用**：不加 `@wraps(func)`，装饰后的函数 `__name__` 会变成 `wrapper`，`__doc__` 也会丢失。`@wraps` 把原函数的元数据复制到 wrapper 上。

### 1.3 带参数的装饰器

```python
import time
from functools import wraps

def retry(max_attempts=3, delay=1):
    """带参数的重试装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts:
                        raise
                    print(f"第 {attempt}/{max_attempts} 次失败：{e}，{delay}s 后重试")
                    time.sleep(delay)
        return wrapper
    return decorator

@retry(max_attempts=3, delay=0.5)
def unstable_api():
    import random
    if random.random() < 0.7:
        raise ConnectionError("网络超时")
    return "成功！"

result = unstable_api()
```

### 1.4 多个装饰器叠加

```python
def log_calls(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"[LOG] 调用 {func.__name__}，参数：{args}, {kwargs}")
        result = func(*args, **kwargs)
        print(f"[LOG] {func.__name__} 返回：{result}")
        return result
    return wrapper

def timer(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"[TIMER] {func.__name__} 耗时 {time.perf_counter()-start:.4f}s")
        return result
    return wrapper

@log_calls
@timer
def add(a, b):
    return a + b

add(3, 5)
# [LOG] 调用 add，参数：(3, 5), {}
# [TIMER] add 耗时 0.0001s
# [LOG] add 返回：8
```

> **装饰器顺序**：从下往上应用。`@log_calls @timer def f():` 等价于 `f = log_calls(timer(f()))`。

## 二、实用装饰器模式

### 2.1 缓存装饰器（memoization）

```python
import functools

def cache(func):
    """简单缓存装饰器"""
    cache_store = {}
    
    @functools.wraps(func)
    def wrapper(*args):
        if args not in cache_store:
            cache_store[args] = func(*args)
        return cache_store[args]
    return wrapper

@cache
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

print(fibonacci(50))  # 12586269025（瞬间返回）
```

> **实际开发**：直接用 `functools.lru_cache` 或 `functools.cache`（Python 3.9+），不需要自己实现。

### 2.2 参数验证装饰器

```python
import functools
import inspect

def validate_types(**type_hints):
    """验证参数类型的装饰器"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            sig = inspect.signature(func)
            params = sig.parameters
            
            # 检查位置参数
            for i, (name, param) in enumerate(params.items()):
                if i < len(args) and name in type_hints:
                    expected = type_hints[name]
                    if not isinstance(args[i], expected):
                        raise TypeError(
                            f"参数 {name} 期望 {expected.__name__}，"
                            f"得到 {type(args[i]).__name__}"
                        )
            
            # 检查关键字参数
            for name, value in kwargs.items():
                if name in type_hints and not isinstance(value, type_hints[name]):
                    raise TypeError(
                        f"参数 {name} 期望 {type_hints[name].__name__}，"
                        f"得到 {type(value).__name__}"
                    )
            
            return func(*args, **kwargs)
        return wrapper
    return decorator

@validate_types(prompt=str, temperature=float)
def call_llm(prompt, temperature):
    print(f"调用 LLM，temperature={temperature}")

call_llm("你好", 0.7)  # ✓
# call_llm(123, 0.7)   # ✗ TypeError
```

### 2.3 限流装饰器

```python
import time
from collections import deque

def rate_limit(max_calls=10, period=1):
    """每分钟最多调用 max_calls 次"""
    def decorator(func):
        timestamps = deque()
        
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()
            # 清理过期时间戳
            while timestamps and timestamps[0] < now - period:
                timestamps.popleft()
            
            if len(timestamps) >= max_calls:
                wait_time = period - (now - timestamps[0])
                raise RuntimeError(
                    f"请求过于频繁，请在 {wait_time:.1f} 秒后重试"
                )
            
            timestamps.append(now)
            return func(*args, **kwargs)
        return wrapper
    return decorator

@rate_limit(max_calls=5, period=60)
def call_api():
    print("API 调用成功")
```

## 三、上下文管理器

### 3.1 类实现上下文管理器

```python
import time

class Timer:
    """计时器上下文管理器"""
    def __init__(self, label="操作"):
        self.label = label
        self.elapsed = None
    
    def __enter__(self):
        self.start = time.perf_counter()
        return self  # __enter__ 的返回值绑定到 as 变量
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.elapsed = time.perf_counter() - self.start
        print(f"[{self.label}] 耗时 {self.elapsed:.4f}s")
        return False  # False = 不吞掉异常
    
    def __repr__(self):
        return f"Timer({self.label}, {self.elapsed:.4f}s)"

# 使用
with Timer("数据处理") as t:
    import time
    time.sleep(0.5)
    result = sum(range(1_000_000))

print(f"结果：{result}")
print(f"Timer 对象：{t}")
```

### 3.2 生成器实现上下文管理器

```python
import contextlib

@contextlib.contextmanager
def database_connection(host, database):
    """模拟数据库连接"""
    print(f"连接到 {host}/{database}")
    try:
        conn = {"host": host, "db": database, "connected": True}
        yield conn  # 这里相当于 with 块
    finally:
        print(f"断开连接 {host}/{database}")
        conn["connected"] = False

# 使用
with database_connection("localhost", "mydb") as conn:
    print(f"连接状态：{conn['connected']}")
    # 执行 SQL...
# 自动断开连接
```

### 3.3 上下文管理器组合

```python
import contextlib

# 同时使用多个上下文管理器
with open("file1.txt") as f1, open("file2.txt") as f2:
    data1 = f1.read()
    data2 = f2.read()

# 或者用 contextlib.ExitStack
from contextlib import ExitStack

files = ["file1.txt", "file2.txt", "file3.txt"]
with ExitStack() as stack:
    fs = [stack.enter_context(open(f)) for f in files]
    # 所有文件自动关闭
```

## 四、AI/LLM 场景实战

### 4.1 延迟日志装饰器

```python
import time
import functools
import logging

logger = logging.getLogger(__name__)

def log_latency(min_ms=100):
    """记录慢操作的装饰器"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.perf_counter()
            result = func(*args, **kwargs)
            elapsed_ms = (time.perf_counter() - start) * 1000
            if elapsed_ms > min_ms:
                logger.warning(
                    f"慢操作: {func.__name__} 耗时 {elapsed_ms:.0f}ms"
                )
            return result
        return wrapper
    return decorator

@log_latency(min_ms=50)
def generate_response(prompt):
    time.sleep(0.1)  # 模拟慢操作
    return "这是回复"

generate_response("你好")
# 输出：[WARNING] 慢操作: generate_response 耗时 100ms
```

### 4.2 带上下文管理的对话会话

```python
import json
from pathlib import Path

class ConversationSession:
    """对话会话上下文管理器，自动保存/加载"""
    
    def __init__(self, filepath="session.json"):
        self.filepath = Path(filepath)
        self.messages = []
    
    def __enter__(self):
        if self.filepath.exists():
            with open(self.filepath, "r", encoding="utf-8") as f:
                self.messages = json.load(f)
            print(f"加载了 {len(self.messages)} 条历史消息")
        return self
    
    def __exit__(self, *args):
        with open(self.filepath, "w", encoding="utf-8") as f:
            json.dump(self.messages, f, ensure_ascii=False, indent=2)
        print(f"保存了 {len(self.messages)} 条消息")
    
    def add(self, role, content):
        self.messages.append({"role": role, "content": content})
    
    def __iter__(self):
        return iter(self.messages)

# 使用
with ConversationSession("chat_session.json") as session:
    session.add("user", "你好")
    session.add("assistant", "你好！有什么可以帮你的？")
    session.add("user", "Python 是什么？")

# 再次使用时自动加载历史
with ConversationSession("chat_session.json") as session:
    print(f"已有 {len(session.messages)} 条消息")
```

## 五、本章小结

| 概念 | 核心要点 |
|------|---------|
| 一等公民 | 函数可赋值、传递、返回 |
| 装饰器 | `@decorator` 语法糖，本质是高阶函数 |
| `@wraps` | 保留原函数的 `__name__` 和 `__doc__` |
| 带参装饰器 | 三层嵌套函数：`decorator(args) -> decorator -> wrapper` |
| 上下文管理器 | `__enter__` / `__exit__` 协议 |
| `@contextmanager` | 用生成器简洁实现上下文管理器 |
| 实用装饰器 | 缓存、重试、限流、日志、参数验证 |
