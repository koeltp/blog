## 一、类与对象基础

### 1.1 定义类

```python
class Dog:
    """狗的类"""
    
    # 类属性（所有实例共享）
    species = "Canis familiaris"
    
    # 构造方法
    def __init__(self, name: str, age: int):
        # 实例属性（每个实例独立）
        self.name = name
        self.age = age
    
    # 实例方法
    def bark(self) -> str:
        return f"{self.name} 说：汪汪！"
    
    def birthday(self):
        self.age += 1
        return f"{self.name} 现在 {self.age} 岁了！"

# 创建实例
dog = Dog("旺财", 3)
print(dog.name)      # 旺财
print(dog.bark())    # 旺财说：汪汪！
dog.birthday()
print(dog.age)       # 4
```

> **`self` 是什么？** `self` 是实例本身的引用。Python 自动将实例作为第一个参数传入方法。你可以把它理解为"这个对象自己"。

### 1.2 类属性 vs 实例属性

```python
class User:
    # 类属性：所有用户共享
    company = "AI Corp"
    user_count = 0
    
    def __init__(self, name):
        self.name = name  # 实例属性：每个用户独立
        User.user_count += 1

u1 = User("Alice")
u2 = User("Bob")

print(u1.company)     # AI Corp（类属性）
print(User.company)   # AI Corp（通过类访问）
print(u1.name)        # Alice（实例属性）
print(User.user_count)  # 2
```

## 二、继承与多态

### 2.1 基本继承

```python
# 父类（基类）
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        return f"{self.name} 发出声音"

# 子类
class Cat(Animal):
    def speak(self):
        return f"{self.name} 说：喵~"

class Dog(Animal):
    def speak(self):
        return f"{self.name} 说：汪汪！"
    
    def fetch(self, item):
        return f"{self.name} 去拿 {item}"

# 使用
cat = Cat("咪咪")
dog = Dog("旺财")

print(cat.speak())  # 咪咪说：喵~
print(dog.speak())  # 旺财说：汪汪！
print(dog.fetch("球"))  # 旺财去拿球
```

### 2.2 super() 调用父类

```python
class BaseLogger:
    def __init__(self, level="INFO"):
        self.level = level
    
    def log(self, message):
        print(f"[{self.level}] {message}")

class FileLogger(BaseLogger):
    def __init__(self, level, filename="log.txt"):
        super().__init__(level)  # 调用父类 __init__
        self.filename = filename
    
    def log(self, message):
        super().log(message)  # 调用父类 log
        with open(self.filename, "a", encoding="utf-8") as f:
            f.write(f"[{self.level}] {message}\n")

logger = FileLogger("DEBUG", "app.log")
logger.log("程序启动")
# 控制台输出：[DEBUG] 程序启动
# 文件写入：[DEBUG] 程序启动
```

### 2.3 多重继承

```python
class Speaks:
    def speak(self):
        return "说话"

class WagsTail:
    def wag_tail(self):
        return "摇尾巴"

# 多重继承
class FriendlyDog(Speaks, WagsTail):
    pass

dog = FriendlyDog()
print(dog.speak())     # 说话
print(dog.wag_tail())  # 摇尾巴
```

> **MRO（方法解析顺序）**：Python 用 C3 线性化算法确定多重继承的方法查找顺序。可以用 `ClassName.__mro__` 查看。

### 2.4 多态

```python
# 不同的对象用相同的方法名
class Bird:
    def speak(self):
        return "叽叽喳喳"

class Robot:
    def speak(self):
        return "哔哔"

def make_it_speak(animal):
    """不关心具体类型，只关心有没有 speak 方法"""
    print(animal.speak())

make_it_speak(Bird())    # 叽叽喳喳
make_it_speak(Robot())   # 哔哔
```

## 三、封装与属性

### 3.1 私有属性

```python
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner          # 公共属性
        self._balance = balance     # 受保护属性（约定不直接访问）
        self.__pin = "1234"         # 私有属性（名称改写）
    
    def deposit(self, amount):
        if amount > 0:
            self._balance += amount
    
    def get_balance(self):
        return self._balance

account = BankAccount("Alice", 1000)
print(account.get_balance())   # 1000
# print(account.__pin)         # ✗ AttributeError
```

> **Python 的"私有"是约定而非强制**：`_` 开头表示"内部使用"，`__` 开头会触发名称改写（变成 `_ClassName__attr`）。Python 哲学是"我们都是成年人"——靠约定而非强制。

### 3.2 property 装饰器

```python
class Temperature:
    def __init__(self, celsius=0):
        self._celsius = celsius
    
    @property
    def celsius(self):
        """获取摄氏温度"""
        return self._celsius
    
    @celsius.setter
    def celsius(self, value):
        if value < -273.15:
            raise ValueError("温度不能低于绝对零度")
        self._celsius = value
    
    @property
    def fahrenheit(self):
        """华氏温度（只读）"""
        return self._celsius * 9/5 + 32

temp = Temperature(25)
print(temp.celsius)    # 25
print(temp.fahrenheit) # 77.0
temp.celsius = 30
print(temp.fahrenheit) # 86.0
```

## 四、魔术方法（Dunder Methods）

魔术方法是 Python 中双下划线开头和结尾的特殊方法，让对象支持内置操作。

### 4.1 常用魔术方法

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __repr__(self):
        """官方字符串表示"""
        return f"Vector({self.x}, {self.y})"
    
    def __str__(self):
        """用户友好的字符串表示"""
        return f"({self.x}, {self.y})"
    
    def __add__(self, other):
        """支持 + 运算符"""
        return Vector(self.x + other.x, self.y + other.y)
    
    def __eq__(self, other):
        """支持 == 运算符"""
        return self.x == other.x and self.y == other.y
    
    def __abs__(self):
        """支持 abs() 内置函数"""
        return (self.x ** 2 + self.y ** 2) ** 0.5
    
    def __len__(self):
        """支持 len() 内置函数"""
        return 2
    
    def __bool__(self):
        """支持 bool() 内置函数"""
        return self.x != 0 or self.y != 0

v1 = Vector(3, 4)
v2 = Vector(1, 2)

print(v1)        # (3, 4)
print(repr(v1))  # Vector(3, 4)
print(v1 + v2)   # (4, 6)
print(v1 == v2)  # False
print(abs(v1))   # 5.0
print(bool(v1))  # True
```

### 4.2 容器魔术方法

```python
class ScoreBoard:
    """支持索引、迭代、长度检查的记分板"""
    
    def __init__(self):
        self._scores = {}
    
    def __setitem__(self, name, score):
        self._scores[name] = score
    
    def __getitem__(self, name):
        return self._scores[name]
    
    def __delitem__(self, name):
        del self._scores[name]
    
    def __contains__(self, name):
        return name in self._scores
    
    def __len__(self):
        return len(self._scores)
    
    def __iter__(self):
        return iter(self._scores.items())
    
    def __repr__(self):
        return f"ScoreBoard({self._scores})"

# 使用
board = ScoreBoard()
board["Alice"] = 95
board["Bob"] = 87

print(board["Alice"])    # 95
print("Bob" in board)     # True
print(len(board))         # 2

for name, score in board:
    print(f"{name}: {score}")
```

## 五、类方法与静态方法

```python
class DateParser:
    def __init__(self, format_str="%Y-%m-%d"):
        self.format_str = format_str
    
    # 实例方法（默认）
    def parse(self, date_str):
        from datetime import datetime
        return datetime.strptime(date_str, self.format_str)
    
    # 类方法：工厂构造函数
    @classmethod
    def iso_format(cls):
        """使用 ISO 格式创建解析器"""
        return cls("%Y-%m-%dT%H:%M:%S")
    
    # 静态方法：工具函数
    @staticmethod
    def is_valid_year(year):
        """验证年份是否合法"""
        return 1 <= year <= 9999

# 使用
parser = DateParser.iso_format()
dt = parser.parse("2024-01-15T10:30:00")

print(DateParser.is_valid_year(2024))  # True
print(DateParser.is_valid_year(0))     # False
```

## 六、ABC：抽象基类

```python
from abc import ABC, abstractmethod

class LLMModel(ABC):
    """LLM 模型抽象基类"""
    
    @abstractmethod
    def generate(self, prompt: str) -> str:
        """生成回复"""
        pass
    
    @abstractmethod
    def chat(self, messages: list) -> str:
        """对话"""
        pass
    
    @property
    @abstractmethod
    def model_name(self) -> str:
        """模型名称"""
        pass

# 具体实现
class FakeLLM(LLMModel):
    @property
    def model_name(self):
        return "fake-llm-v1"
    
    def generate(self, prompt):
        return f"[Fake] 对 '{prompt}' 的回复"
    
    def chat(self, messages):
        return f"[Fake] 对话回复，共 {len(messages)} 条消息"

model = FakeLLM()
print(model.generate("你好"))  # [Fake] 对 '你好' 的回复
```

> **为什么需要 ABC？** LangChain 中的所有组件（LLM、Chain、Tool、Memory）都是基于抽象基类设计的。理解 ABC 能帮你读懂 LangChain 的源码架构。

## 七、AI/LLM 场景实战

### 7.1 构建简单的消息类

```python
from enum import Enum
from dataclasses import dataclass, field
from typing import Optional

class Role(Enum):
    SYSTEM = "system"
    USER = "user"
    ASSISTANT = "assistant"

@dataclass
class Message:
    """聊天消息"""
    role: Role
    content: str
    tokens: Optional[int] = None
    
    def __repr__(self):
        return f"Message({self.role.value}, {self.content[:30]}...)"

@dataclass
class Conversation:
    """对话会话"""
    messages: list[Message] = field(default_factory=list)
    
    def add_user_message(self, content: str):
        self.messages.append(Message(Role.USER, content))
    
    def add_assistant_message(self, content: str):
        self.messages.append(Message(Role.ASSISTANT, content))
    
    def get_context(self, max_messages: int = 10) -> list[dict]:
        """获取最近 N 条消息，格式化为 API 需要的格式"""
        return [
            {"role": msg.role.value, "content": msg.content}
            for msg in self.messages[-max_messages:]
        ]

# 使用
conv = Conversation()
conv.add_user_message("Python 是什么？")
conv.add_assistant_message("Python 是一门编程语言。")
conv.add_user_message("能举个例子吗？")
conv.add_assistant_message("当然，让我给你写一个 Hello World。")

print(conv.get_context(max_messages=2))
# [{'role': 'user', 'content': '能举个例子吗？'}, 
#  {'role': 'assistant', 'content': '当然，让我给你写一个 Hello World。'}]
```

### 7.2 为什么 dataclass 重要

```python
# 传统写法
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __repr__(self):
        return f"Point({self.x}, {self.y})"
    
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

# dataclass 写法（等价但简洁得多）
from dataclasses import dataclass

@dataclass
class Point:
    x: float
    y: float

# 自动生成 __init__, __repr__, __eq__, __hash__
p = Point(1.0, 2.0)
print(p)           # Point(1.0, 2.0)
print(p == Point(1.0, 2.0))  # True
```

> **dataclass 在 AI 开发中的应用**：LangChain 中的 Chain、Tool、Memory 等核心数据结构大量使用 dataclass。它是 Python 3.7+ 引入的语法糖，大幅减少样板代码。

## 八、本章小结

| 概念 | 说明 |
|------|------|
| 类与对象 | `class` 定义模板，`__init__` 构造实例 |
| self | 实例自身的引用 |
| 继承 | `class Child(Parent)`，`super()` 调用父类 |
| 封装 | `_` 受保护，`__` 私有（名称改写） |
| property | `@property` 装饰器实现 getter/setter |
| 魔术方法 | `__repr__`, `__add__`, `__len__` 等 |
| 类方法 | `@classmethod`，工厂构造 |
| 静态方法 | `@staticmethod`，工具函数 |
| ABC | `@abstractmethod` 定义接口 |
| dataclass | 自动生成 `__init__`, `__repr__` 等 |
